      else if (event.key === '-' || event.key === '_') { event.preventDefault(); zoomAt(state.scale / 1.2); }
      else if (event.key === '0') { event.preventDefault(); state.chosen = true; apply(fitScale()); }

    var drag = null, touches = {}, pinch = null;
    function distance() {
      var points = Object.keys(touches).map(function (k) { return touches[k]; });
      return points.length < 2 ? 0 : Math.hypot(points[0].x - points[1].x, points[0].y - points[1].y);
    }
    function middle() {
      var points = Object.keys(touches).map(function (k) { return touches[k]; });
      return { x: (points[0].x + points[1].x) / 2, y: (points[0].y + points[1].y) / 2 };
    }
    canvas.addEventListener('pointerdown', function (event) {
      if (event.pointerType === 'touch') {
        touches[event.pointerId] = { x: event.clientX, y: event.clientY };
        if (Object.keys(touches).length === 2) { pinch = { distance: distance(), scale: state.scale }; }
        return;
      }
      if (event.button !== 0 || event.target.closest('.node, .anode, button, a')) { return; }
      drag = { x: event.clientX, y: event.clientY, left: canvas.scrollLeft, top: canvas.scrollTop, moved: false };
    });
    canvas.addEventListener('pointermove', function (event) {
      if (event.pointerType === 'touch') {
        if (!touches[event.pointerId]) { return; }
        touches[event.pointerId] = { x: event.clientX, y: event.clientY };
        if (pinch && pinch.distance) {
          var center = middle();
          zoomAt(pinch.scale * distance() / pinch.distance, center.x, center.y);
        }
        return;
      }
      if (!drag) { return; }
      var dx = event.clientX - drag.x, dy = event.clientY - drag.y;
      if (!drag.moved && Math.abs(dx) + Math.abs(dy) > 4) { drag.moved = true; canvas.classList.add('dragging'); }
      if (drag.moved) { canvas.scrollLeft = drag.left - dx; canvas.scrollTop = drag.top - dy; }
    });
    function end(event) {
      if (event && event.pointerType === 'touch') {
        delete touches[event.pointerId];
        if (Object.keys(touches).length < 2) { pinch = null; }
        return;
      }
      if (drag) { canvas.classList.remove('dragging'); drag = null; }
    }
    canvas.addEventListener('pointerup', end);
    canvas.addEventListener('pointerleave', end);
    canvas.addEventListener('pointercancel', end);
    return { settle: settle, zoomAt: zoomAt, state: state, center: function (element) {
      if (!element || !element.getBBox) { return; }
      var box = element.getBBox();
      canvas.scrollTo({ left: Math.max(0, box.x * state.scale - canvas.clientWidth / 2 + box.width * state.scale / 2),
        top: Math.max(0, box.y * state.scale - canvas.clientHeight / 2 + box.height * state.scale / 2), behavior: 'smooth' });
    } };
  }

      (map[next] || []).forEach(function (item) { stack.push(item); });
    }
    return seen;

(function () {
  var root = document.documentElement;

  var scope = document.getElementById('plan-export-scope');

  function capture() {
    return {
      title: document.title,
      scope: root.getAttribute('data-print-scope'),
      printing: body.getAttribute('data-printing'),
      focus: document.activeElement
    };
  }

  function restoreAttribute(element, name, value) {
    if (value === null) { element.removeAttribute(name); }
    else { element.setAttribute(name, value); }
  }

  function restore() {
    if (!active) { return; }
    var saved = active.saved;
    active = null;
    document.title = saved.title;
    restoreAttribute(root, 'data-print-scope', saved.scope);
    restoreAttribute(body, 'data-printing', saved.printing);
    if (saved.focus && typeof saved.focus.focus === 'function') { saved.focus.focus(); }
  }

  function apply(scopeName) {
    root.setAttribute('data-print-scope', scopeName);
    body.setAttribute('data-printing', 'true');
    document.title = active.saved.title + ' — ' + scopeName;
  }

  window.addEventListener('beforeprint', function () {
    if (!active) { active = { scope: 'both', saved: capture() }; }
    apply(active.scope);
  });

  window.addEventListener('afterprint', restore);

  button.addEventListener('click', function (event) {
    if (event && event.preventDefault) { event.preventDefault(); }

    var selected = scope.value;
    if (selected !== 'plan' && selected !== 'progress' && selected !== 'both') { selected = 'both'; }
